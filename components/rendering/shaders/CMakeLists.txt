add_library(rendering_shaders INTERFACE)
ac_component(rendering_shaders rendering.shaders)
file(GLOB_RECURSE engine_shader_inputs CONFIGURE_DEPENDS "${ENGINE_SHADER_ROOT}/*.slang")
set_property(TARGET rendering_shaders PROPERTY ENGINE_SHADER_INPUTS "${engine_shader_inputs}")
if(ENGINE_DESKTOP)
    include("${PROJECT_SOURCE_DIR}/cmake/shaders.cmake")
else()
    add_custom_target(engine_shaders)
endif()
foreach(shader IN LISTS engine_shader_inputs)
    ac_asset(rendering_shaders "${shader}" engine_shaders)
endforeach()
ac_asset(rendering_shaders resources/programs.json engine_shaders)
