cmake_minimum_required(VERSION 4.4.5)
project(engine VERSION 1.1.4 LANGUAGES CXX)
include(GNUInstallDirs)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "23.1.5" OR
   (WIN32 AND NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC"))
    message(FATAL_ERROR "Engine requires LLVM Clang 23.1.5 (clang-cl on Windows)")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(ENGINE_DESKTOP "Build GLFW, Vulkan and Dear ImGui desktop application" ON)
option(ENGINE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENGINE_GPU_TESTS "Register display/GPU/validation-layer integration tests with CTest" OFF)
enable_testing()
include(cmake/archcheck.cmake)
if(ENGINE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "Combined address/undefined sanitizers need Clang or GCC")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/dependencies.cmake)
if(ENGINE_DESKTOP)
    include(cmake/imgui.cmake)
endif()
add_subdirectory(components/foundation/geometry)
add_subdirectory(components/foundation/parallel)
add_subdirectory(components/motion/camera)
add_subdirectory(components/motion/character)
add_subdirectory(components/motion/path)
add_subdirectory(components/interaction/input)
add_subdirectory(components/platform/window_state)
add_subdirectory(components/platform/filesystem)
add_subdirectory(components/platform/process)
add_subdirectory(components/platform/window)
add_subdirectory(components/world/model)
add_subdirectory(components/world/editing)
add_subdirectory(components/world/storage)
add_subdirectory(components/world/collision)
add_subdirectory(components/rendering/settings)
add_subdirectory(components/rendering/configuration)
set(ENGINE_SHADER_ROOT "${PROJECT_SOURCE_DIR}/components/rendering/shaders/resources/shaders")
add_subdirectory(components/rendering/shaders)
add_subdirectory(components/rendering/vulkan)
add_subdirectory(components/gui/frontend)
add_subdirectory(apps/sandbox)
add_subdirectory(apps/benchmark)
if(ENGINE_DESKTOP)
    add_test(NAME installed_engine_test COMMAND "${CMAKE_COMMAND}"
        "-DBUILD_DIR=${CMAKE_BINARY_DIR}" "-DEXECUTABLE_SUFFIX=${CMAKE_EXECUTABLE_SUFFIX}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/installed_engine.cmake")
    ac_package_test(installed_engine_test)
endif()
ac_export_facts("${CMAKE_BINARY_DIR}/architecture.facts")
