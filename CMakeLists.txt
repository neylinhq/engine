cmake_minimum_required(VERSION 3.27)
project(engine VERSION 1.1.4 LANGUAGES CXX)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENGINE_DESKTOP "Build GLFW, Vulkan and Dear ImGui desktop application" ON)
option(ENGINE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENGINE_GPU_TESTS "Register display/GPU/validation-layer integration tests with CTest" OFF)
enable_testing()
include(cmake/architecture.cmake)
if(ENGINE_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "Combined address/undefined sanitizers need Clang or GCC")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Boost 1.83 CONFIG REQUIRED COMPONENTS json)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
ac_external(OpenSSL::Crypto external.openssl "${OPENSSL_VERSION}" "${OPENSSL_INCLUDE_DIR}/openssl")
get_target_property(boost_include_roots Boost::headers INTERFACE_INCLUDE_DIRECTORIES)
set(boost_architecture_roots "")
foreach(directory IN LISTS boost_include_roots)
    list(APPEND boost_architecture_roots "${directory}/boost")
endforeach()
ac_external(Boost::headers external.boost "${Boost_VERSION_STRING}" ${boost_architecture_roots})
ac_external(Boost::json external.boost "${Boost_VERSION_STRING}" ${boost_architecture_roots})
if(ENGINE_DESKTOP)
    find_package(Vulkan 1.3 REQUIRED)
    ac_external(Vulkan::Vulkan external.vulkan "${Vulkan_VERSION}"
        "${Vulkan_INCLUDE_DIR}/vulkan" "${Vulkan_INCLUDE_DIR}/vk_video")
    find_package(glfw3 3.4 CONFIG REQUIRED)
    get_target_property(glfw_roots glfw INTERFACE_INCLUDE_DIRECTORIES)
    set(glfw_architecture_roots "")
    foreach(directory IN LISTS glfw_roots)
        list(APPEND glfw_architecture_roots "${directory}/GLFW")
    endforeach()
    ac_external(glfw external.glfw "3.4" ${glfw_architecture_roots})
    include(cmake/imgui.cmake)
    include(cmake/shaders.cmake)
endif()
add_subdirectory(components/foundation/geometry)
add_subdirectory(components/foundation/parallel)
add_subdirectory(components/motion/camera)
add_subdirectory(components/motion/path)
add_subdirectory(components/interaction/input)
add_subdirectory(components/platform/window_state)
add_subdirectory(components/platform/process)
add_subdirectory(components/platform/window)
add_subdirectory(components/world/model)
add_subdirectory(components/world/editing)
add_subdirectory(components/world/storage)
add_subdirectory(components/rendering/settings)
add_subdirectory(components/rendering/configuration)
add_subdirectory(components/rendering/shaders)
add_subdirectory(components/rendering/vulkan)
add_subdirectory(components/gui/frontend)
add_subdirectory(apps/sandbox)
add_subdirectory(apps/benchmark)
ac_export_facts("${CMAKE_BINARY_DIR}/architecture.facts")
